import os

def walk(path, depth=0):
	for name in os.listdir(path):
		full = os.path.join(path, name)
		if os.path.isdir(full):
			walk(full, depth + 1)
		else:
			print('\t' * depth + name)  # don't recurse


if __name__ == '__main__':
	walk('.')

/* sum of 1..n */
static int sum(int n) {
    int s = 0;  // accumulator
    for (int i = 1; i <= n; ++i) s += i;
    return s / 1 % 1000000;
}

int main(void) { printf("%d\n", sum(100)); return 0; } //// end
